### scripts\04c_run_classical_premise_test.py :: censoring_km_training
### signature: censoring_km_training(y_train: np.ndarray) -> Tuple[np.ndarray, np.ndarray]
### source_sha256: e1490812955df2681ae7280317be11d4c56b91792747a6b24dcf80635404a8eb

def censoring_km_training(
    y_train: np.ndarray,
) -> Tuple[np.ndarray, np.ndarray]:
    times = np.asarray(y_train["time"], dtype=float)
    original_event = np.asarray(y_train["event"], dtype=bool)
    censor_event = ~original_event

    unique_times = np.sort(np.unique(times))
    surv = 1.0
    out_times = []
    out_surv = []

    for t in unique_times:
        at_risk = int(np.sum(times >= t))
        d_censor = int(np.sum((times == t) & censor_event))

        if at_risk > 0 and d_censor > 0:
            surv *= 1.0 - d_censor / at_risk

        out_times.append(float(t))
        out_surv.append(float(surv))

    return np.asarray(out_times), np.asarray(out_surv)


### scripts\04c_run_classical_premise_test.py :: ipcw_brier_score
### signature: ipcw_brier_score(y_train: np.ndarray, y_test: np.ndarray, survival_prob: np.ndarray, eval_times: np.ndarray) -> np.ndarray
### source_sha256: 4bd9dafe3d426a4405f7e0928c97126659f9e6e328833077eb5fdb5052fd4c90

def ipcw_brier_score(
    y_train: np.ndarray,
    y_test: np.ndarray,
    survival_prob: np.ndarray,
    eval_times: np.ndarray,
) -> np.ndarray:
    km_times, km_surv = censoring_km_training(y_train)

    test_time = np.asarray(y_test["time"], dtype=float)
    test_event = np.asarray(y_test["event"], dtype=bool)

    survival_prob = np.asarray(survival_prob, dtype=float)
    eval_times = np.asarray(eval_times, dtype=float)

    if survival_prob.shape != (len(y_test), len(eval_times)):
        raise RuntimeError("Brier survival-probability shape mismatch.")

    scores = []

    for j, t in enumerate(eval_times):
        G_t = km_step_value(km_times, km_surv, t, left_limit=False)
        if G_t <= 1e-8:
            raise RuntimeError("Training censoring survival too small at IBS time.")

        weighted_error = np.zeros(len(y_test), dtype=float)

        for i in range(len(y_test)):
            Ti = float(test_time[i])
            di = bool(test_event[i])
            S = float(survival_prob[i, j])

            if Ti <= t and di:
                G_Tminus = km_step_value(
                    km_times,
                    km_surv,
                    Ti,
                    left_limit=True,
                )
                if G_Tminus <= 1e-8:
                    raise RuntimeError(
                        "Training censoring survival too small at test event time."
                    )
                weighted_error[i] = (S ** 2) / G_Tminus

            elif Ti > t:
                weighted_error[i] = ((1.0 - S) ** 2) / G_t

            else:
                # Censored at/before t: zero IPCW contribution.
                weighted_error[i] = 0.0

        scores.append(float(np.mean(weighted_error)))

    return np.asarray(scores, dtype=float)


### scripts\04c_run_classical_premise_test.py :: choose_ibs_grid
### signature: choose_ibs_grid(y_train: np.ndarray, y_test: np.ndarray) -> np.ndarray
### source_sha256: f4619186849aba855585a6b6f1945437d8e2e761fd104e98a5b024d69866e2e1

def choose_ibs_grid(
    y_train: np.ndarray,
    y_test: np.ndarray,
) -> np.ndarray:
    train_time = np.asarray(y_train["time"], dtype=float)
    train_event = np.asarray(y_train["event"], dtype=bool)
    test_time = np.asarray(y_test["time"], dtype=float)

    event_times = train_time[train_event]
    if len(event_times) < 2:
        raise RuntimeError("Too few training events for frozen IBS grid.")

    lo = float(np.quantile(event_times, IBS_LOW_Q))
    hi = float(np.quantile(event_times, IBS_HIGH_Q))

    if not np.isfinite(lo) or not np.isfinite(hi) or hi <= lo:
        raise RuntimeError("Invalid frozen IBS event-time quantiles.")

    grid = np.linspace(lo, hi, IBS_GRID_POINTS)

    lower_support = max(
        float(np.min(train_time)),
        float(np.min(test_time)),
    )
    upper_support = min(
        float(np.max(train_time)),
        float(np.max(test_time)),
    )

    grid = grid[
        (grid > lower_support)
        & (grid < np.nextafter(upper_support, 0.0))
    ]

    if len(grid) < IBS_MIN_POINTS:
        raise RuntimeError(
            f"Only {len(grid)} supported IBS points; "
            f"minimum={IBS_MIN_POINTS}."
        )

    return grid


### scripts\04c_run_classical_premise_test.py :: integrated_brier_custom
### signature: integrated_brier_custom(y_train: np.ndarray, y_test: np.ndarray, survival_prob: np.ndarray, eval_times: np.ndarray) -> float
### source_sha256: 4bcfd0e1e54881b4b732633e2c86af9862770acae9c9ae65d2602d6786ac8252

def integrated_brier_custom(
    y_train: np.ndarray,
    y_test: np.ndarray,
    survival_prob: np.ndarray,
    eval_times: np.ndarray,
) -> float:
    bs = ipcw_brier_score(
        y_train,
        y_test,
        survival_prob,
        eval_times,
    )
    span = float(eval_times[-1] - eval_times[0])
    if span <= 0:
        raise RuntimeError("Invalid IBS integration span.")
    return float(np.trapz(bs, eval_times) / span)


### scripts\04c_run_classical_premise_test.py :: weighted_mean_ibs
### signature: weighted_mean_ibs(ibs: pd.DataFrame, model: str) -> float
### source_sha256: 589cb2f1fdddf799c356db2cc2b97024b7ce1eb0ef04a5230e65c8b850d3a7b7

def weighted_mean_ibs(
    ibs: pd.DataFrame,
    model: str,
) -> float:
    part = ibs[ibs["model"] == model].copy()
    weights = part["n_test"].to_numpy(dtype=float)
    values = part["ibs"].to_numpy(dtype=float)

    if not np.isfinite(values).all():
        raise RuntimeError(f"{model}: non-finite IBS values.")

    return float(np.average(values, weights=weights))

def _breslow_survival_from_fixed_risk(
    train_time: np.ndarray,
    train_event: np.ndarray,
    train_risk: np.ndarray,
    test_risk: np.ndarray,
    eval_times: np.ndarray,
) -> np.ndarray:
    """
    Target-training-only Breslow baseline recalibration for a fixed risk score.

    No coefficient, slope, intercept, or rank score is refit.
    This secondary IBS adapter places all ranking models on the TARGET time scale.
    """
    train_time = np.asarray(
        train_time,
        dtype=float,
    )
    train_event = np.asarray(
        train_event,
        dtype=bool,
    )
    train_risk = np.asarray(
        train_risk,
        dtype=float,
    )
    test_risk = np.asarray(
        test_risk,
        dtype=float,
    )
    eval_times = np.asarray(
        eval_times,
        dtype=float,
    )

    if not (
        np.isfinite(train_time).all()
        and np.isfinite(train_risk).all()
        and np.isfinite(test_risk).all()
    ):
        raise RuntimeError(
            "Non-finite Breslow input."
        )

    eta = np.clip(
        train_risk,
        -50,
        50,
    )
    exp_eta = np.exp(eta)

    event_times = np.unique(
        train_time[train_event]
    )

    increments = []

    for t in event_times:
        d = int(
            np.sum(
                train_event
                & (train_time == t)
            )
        )
        risk_sum = float(
            np.sum(
                exp_eta[
                    train_time >= t
                ]
            )
        )

        if (
            risk_sum <= 0
            or not np.isfinite(
                risk_sum
            )
        ):
            raise RuntimeError(
                "Invalid Breslow risk-set sum."
            )

        increments.append(
            d / risk_sum
        )

    cum = np.cumsum(
        np.asarray(
            increments,
            dtype=float,
        )
    )

    idx = np.searchsorted(
        event_times,
        eval_times,
        side="right",
    ) - 1

    H0 = np.zeros(
        len(eval_times),
        dtype=float,
    )

    valid = idx >= 0
    H0[valid] = cum[
        idx[valid]
    ]

    multiplier = np.exp(
        np.clip(
            test_risk,
            -50,
            50,
        )
    )

    return np.exp(
        -multiplier[:, None]
        * H0[None, :]
    )
